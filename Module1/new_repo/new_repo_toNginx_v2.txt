#Отдельная конфигурация для nginx
slab@slab:/var/www/html/repo$ cat /etc/nginx/sites-available/otus-repo
#nginx
server {
    listen 80;
    server_name _;

    root /var/www/html;
    index index.html;

    location /repo/ {
        autoindex on;
        try_files $uri $uri/ =404;
    }

    location /repo/conf/ {
        deny all;
    }

    location /repo/db/ {
        deny all;
    }
}

#Включили конфигурацию, "сделали симлинк"
slab@slab:/var/www/html/repo$ sudo ln -s /etc/nginx/sites-enabled/ /etc/nginx/sites-available/
default    otus-repo
slab@slab:/var/www/html/repo$ sudo ln -s  /etc/nginx/sites-available/otus-repo /etc/nginx/sites-enabled/otus-repo
